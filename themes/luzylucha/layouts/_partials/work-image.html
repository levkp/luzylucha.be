{{- /* Returns the image resource for a work, or nothing. Uses the front matter "image" (a file in the bundle) when set, otherwise the first image in the page bundle. @context {page} The work page. @returns {resource} */ -}} {{- $img := "" }} {{- if .IsPage }} {{- with .Params.image }}{{ $img = $.Resources.Get . }}{{ end }} {{- if not $img }}{{ $img = index (.Resources.ByType "image") 0 }}{{ end }} {{- end }} {{- return $img }}
